\section{ From Signals to Series}
\subsection{The Fourier Series}
Let $f(t)$ be a periodic function with period $T$. Its Fourier series representation is
\begin{equation}
    f(t)=a_0+\sum_{n=1}^\infty \left[ a_n \cos(\frac{2\pi nt}{T}) + b_n \sin(\frac{2\pi nt}{T})  \right]
    \label{eq:1}
\end{equation}
where the coefficients are given by
\begin{equation}
    a_0=\frac{2}{T}\int _0 ^T f(t) \hspace{10pt}
     a_n=\frac{2}{T}\int _0 ^T f(t) \cos(\frac{2\pi nt}{T})\,dt ,\hspace{10pt}
      b_n=\frac{2}{T}\int _0 ^T f(t) \sin(\frac{2\pi nt}{T})\,dt ,
\end{equation}
\textcolor{blue}{An even function contains only cosine terms ($b_n$ = 0 for all n)}, whereas \textcolor{red}{an odd function contains
only sine terms ($a_n$ = 0 for all n)}.\footnote{A function that is neither even nor odd generally requires both families of terms.}
\subsection{ A Simple Example: the Square Wave}
For the odd square wave $$f(t) = sgn(\sin(2\pi t/T))$$ of amplitude 1,
\begin{align}
    b_n &= \frac{2}{T}\int _0 ^T f(t) \sin(\frac{2\pi nt}{T})\,dt , \notag  \\ 
    &=\frac{4}{T}\int _0 ^T/2 f(t) \sin(\frac{2\pi nt}{T})\,dt , \notag \\
    &= 
    \begin{cases} 
        \frac{4}{n\pi} & , \; n\; \text{odd} \\
        0& , \; n\; \text{even}
    \end{cases}
    \label{eq:three}
\end{align}
so that the series contains only odd harmonics. A concise derivation is available from the \href{https://mathworld.wolfram.com/FourierSeriesSquareWave.html}{Square
Wave entry on MathWorld}, and the underlying convergence theory is due to Dirichlet.\cite{dirichlet1829}

\section{Approximating a Square Wave from Harmonics}
\begin{wrapfigure}{r}{0.36\linewidth}
    \centering

    \caption{The ideal square}
    \label{fig:ideal}

Figure \ref{fig:sq} shows the ideal square wave that Equation \eqref{eq:three} at
tempts to reconstruct from a finite number of harmonics. Trun
cating the series in Equation \eqref{eq:1} at N terms gives the partial
sum
\begin{equation}
    s_N(t)=\sum_{n=1}^N \frac{4}{n\pi} \sin \left( \frac{2n\pi t }{T}\right)
    \label{eq:4}
\end{equation}

As $N$ increases, $s_N(t)$ converges to $f(t)$ everywhere except at
the jump discontinuities, near which an overshoot of approxi
mately 9\% persists no matter how large $N$ becomes.

\begin{proposition}[Gibbs Overshoot]
    \label{prop:G}
    For the square wave par
tial sum $s_N(t)$ in Equation \eqref{eq:4}, the overshoot near a jump dis
continuity does not vanish as $N\to \infty$; instead,.
\begin{equation}
    \lim _{N \to \infty} s_N \left ( \frac{T}{2N}\right) \approx 1.1789
\end{equation}
an overshoot of about 9\% above the jump height of 1.
\end{proposition}

\begin{proof}
    Write $s_N(t)$ as an integral of the Dirichlet kernel and evaluate at $t = T/(2N)$; the result
reduces to the sine integral $Si(\pi)$, which exceeds $\pi/2$ by a fixed amount independent of $N$
\end{proof} 
See Proposition \ref{prop:G} for the precise statement of this bound \cite{oppenheim1997}
\section{Where Fourier Analysis Is Used}
Fourier analysis appears across a wide range of engineering disciplines:
\begin{enumerate}[label=$\oplus$]
    \item Audio and music
    \begin{enumerate}[label=$\oplus$]
        \item MP3 and AAC compression discard inaudible high-frequency content;
        \item  graphic equalizers boost or cut specific frequency bands.
    \end{enumerate}
    \item  Image processing
        \begin{enumerate}[label=$\oplus$]
        \item JPEG compression uses the related discrete cosine transform
        \item  edge detection filters operate on high spatial frequencies.
        \end{enumerate}
    \item  Communications
    \begin{enumerate}[label=$\oplus$]
        \item modulation shifts a signal’s spectrum to a carrier frequency
        \item  ecommon multiplexing schemes include:
        \begin{itemize}[label=$\triangleright$]
                \item frequency-division multiplexing;
                \item   orthogonal frequency-division multiplexing;
              \item  wavelength-division multiplexing
        \end{itemize}
    \end{enumerate}
    
\end{enumerate}
